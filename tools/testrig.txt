# Test contraption for Gauge, built in front of a player at 5 -60 1 looking south.
# Use with: ./gradlew runClient "-Pdevshots=3,-60,3;6,-60,3;7,-60,3;8,-60,3;9,-60,3" (chest, then focus targets)
fill -2 -60 2 14 -55 16 air
# lever -> dust line -> repeater -> dust
setblock 0 -60 5 lever[face=floor,powered=true]
fill 1 -60 5 6 -60 5 redstone_wire
setblock 7 -60 5 repeater[facing=west,delay=2]
fill 8 -60 5 9 -60 5 redstone_wire
# observer clock
setblock 2 -60 8 observer[facing=east]
setblock 3 -60 8 observer[facing=west]
setblock 4 -60 8 redstone_wire
# hopper -> chest <- comparator -> dust (within reach of the player)
setblock 3 -60 3 chest
setblock 3 -59 3 hopper[facing=down]
setblock 2 -60 3 comparator[facing=east]
setblock 1 -60 3 redstone_wire
item replace block 3 -60 3 container.0 with stone 64
item replace block 3 -60 3 container.1 with stone 64
item replace block 3 -60 3 container.2 with stone 64
item replace block 3 -60 3 container.3 with stone 64
item replace block 3 -60 3 container.4 with stone 64
item replace block 3 -60 3 container.5 with stone 64
item replace block 3 -60 3 container.6 with stone 64
item replace block 3 -60 3 container.7 with stone 64
item replace block 3 -60 3 container.8 with stone 64
item replace block 3 -60 3 container.9 with stone 64
# composter and a torch-powered block
setblock 8 -60 8 composter[level=5]
setblock 10 -60 8 redstone_torch
setblock 10 -59 8 stone
# note blocks (harp on grass, bell on gold, bass drum on stone) and a jukebox
setblock 6 -60 3 note_block[instrument=harp,note=6]
setblock 7 -61 3 gold_block
setblock 7 -60 3 note_block[instrument=bell,note=12]
setblock 9 -61 3 stone
setblock 9 -60 3 note_block[instrument=basedrum,note=3]
setblock 8 -60 3 jukebox
# the disc goes in later (see README) so the song starts while the client watches
# player
gamemode creative GaugeTester
tp GaugeTester 5 -60 1 0 30
